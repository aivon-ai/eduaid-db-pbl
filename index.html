// 필수값 비어 있음, 계약번호 중복, 금액 이상을 찾는다
function findContractIssues_(rows) {
  var issues = [], seen = {};
  rows.forEach(function (r, i) {
    var line = i + 2, no = String(r[0]).trim();
    if (!no) issues.push([line, '계약번호', '비어 있음']);
    else if (seen[no]) issues.push([line, '계약번호', '중복(' + seen[no] + '행과 동일)']);
    else seen[no] = line;
    if (!String(r[2]).trim()) issues.push([line, '수요기관', '비어 있음']);
    if (!(Number(r[9]) > 0)) issues.push([line, '계약금액', '0 이하 또는 숫자가 아님']);
  });
  return issues;
}
